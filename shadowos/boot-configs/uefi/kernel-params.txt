# UEFI Kernel Parameters
# Optimized kernel boot parameters for UEFI systems

root=/dev/mapper/shadowos-root

# Console and output
console=tty0

# Security parameters

security=apparmor

# TPM and secure boot
tpm=crb

# Kernel hardening
kernel.kptr_restrict=2
kernel.dmesg_restrict=1

# Memory protection
slub_debug=F
slab_nomerge

# IOMMU and virtualization
iommu=force
amd_iommu=force
intel_iommu=force

# Disable unnecessary features
nowatchdog
nokaslr

# Performance parameters
preempt=voluntary
